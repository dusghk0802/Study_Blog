//아래와 같이 EMPLOYEE 테이블을 생성
//( EMPLOYEE_ID는 자동 증가, PRIMARY KEY, NAME는 NOT NULL)

--Name Null Type
-------- ------------- ------------
--EMPLOYEE_ID NUMBER(7)
--NAME VARCHAR2(25)
--ADDRESS VARCHAR2(25)
--DEPTNO NUMBER(2)
--PHONE_NUMBER VARCHAR2(20)

CREATE TABLE EMPLOYEE
   (EMPLOYEE_ID NUMBER(7) PRIMARY KEY,
    NAME VARCHAR2(25) NOT NULL,
    ADDRESS VARCHAR2(25),
    DEPTNO NUMBER(2),
    PHONE_NUMBER VARCHAR2(20));
    
CREATE SEQUENCE EMPLOYEE_SEQ
START WITH 1
INCREMENT BY 1;
