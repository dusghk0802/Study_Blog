# ==========================================
# 01. 출력 리다이렉션 >
# > 는 명령어의 표준 출력을 파일에 저장하며, 기존 내용이 있으면 덮어쓴다.
# ==========================================

user@ubuntu:~$ mkdir redirect-test
user@ubuntu:~$ cd redirect-test/

user@ubuntu:~/redirect-test$ echo "hello word" > greetings

user@ubuntu:~/redirect-test$ ls -l
total 4
-rw-rw-r-- 1 user user 11 Sep 15 09:31 greetings

user@ubuntu:~/redirect-test$ cat greetings
hello word


# ==========================================
# 02. > 를 이용한 파일 내용 덮어쓰기
# ==========================================

user@ubuntu:~/redirect-test$ echo "hi~~~" > greetings

user@ubuntu:~/redirect-test$ cat greetings
hi~~~


# ==========================================
# 03. >> 를 이용한 파일 내용 추가
# >> 는 기존 내용을 유지하면서 명령어의 표준 출력을 파일 끝에 추가한다.
# ==========================================

user@ubuntu:~/redirect-test$ echo "hi here" >> greetings

user@ubuntu:~/redirect-test$ cat greetings
hi~~~
hi here

user@ubuntu:~/redirect-test$ echo "how are you?" >> greetings

user@ubuntu:~/redirect-test$ cat greetings
hi~~~
hi here
how are you?


# ==========================================
# 04. 표준 에러 리다이렉션 2>
# 2> 는 명령 실행 중 발생한 표준 에러(stderr)를 파일에 저장한다.
# ==========================================

user@ubuntu:~/redirect-test$ this is error command 2> errmsg

user@ubuntu:~/redirect-test$ cat errmsg
Command 'this' not found, did you mean:
  command 'thin' from deb thin (1.8.1-2ubuntu1)
Try: sudo apt install <deb name>


# ==========================================
# 05. 표준 출력과 표준 에러 분리

user@ubuntu:~/redirect-test$ ping -c 1 www.google.com > stdout 2> stderror

user@ubuntu:~/redirect-test$ echo $?

user@ubuntu:~/redirect-test$ echo $?

# 7. tar 압축 파일 생성

# ==================================================
# 16. systemd 서비스 자동 시작 비활성화
# disable은 부팅 시 자동 시작 설정을 해제하며, 현재 실행 중인 서비스를 자동으로 중지하지는 않는다.
# ==================================================

user@ubuntu:~$ sudo systemctl disable myservice
Removed "/etc/systemd/system/multi-user.target.wants/myservice.service".

user@ubuntu:~$ sudo systemctl status myservice
○ myservice.service - My Test Service
     Loaded: loaded (/etc/systemd/system/myservice.service; disabled; preset: enabled)
     Active: inactive (dead)


# ==================================================
# 17. systemd 실습 파일 삭제
# ==================================================

user@ubuntu:~$ sudo rm /etc/systemd/system/myservice.service
user@ubuntu:~$ sudo rm /tmp/myservice.sh
user@ubuntu:~$ sudo rm /tmp/myservice.log

# 압축 실습용 디렉터리 생성
user@ubuntu:~$ mkdir practice_exam
user@ubuntu:~$ cd practice_exam/
# 여러 개의 빈 실습 파일 생성
user@ubuntu:~/practice_exam$ touch apple.txt banana.txt orange.log
user@ubuntu:~/practice_exam$ ls
apple.txt  banana.txt  orange.log
user@ubuntu:~/practice_exam$ cd
# 디렉터리 전체를 gzip 방식의 tar 파일로 압축
user@ubuntu:~$ tar -cvzf practice_exam.tar.gz practice_exam/
practice_exam/
practice_exam/apple.txt
practice_exam/banana.txt
practice_exam/orange.log
# 압축 해제 결과를 저장할 디렉터리 생성
user@ubuntu:~$ mkdir /tmp/exam_restore
# -C 옵션으로 지정한 디렉터리에 압축 해제
user@ubuntu:~$ tar -xvzf practice_exam.tar.gz -C /tmp/exam_restore
